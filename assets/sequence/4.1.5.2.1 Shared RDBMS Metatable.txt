participant "Distributed Concurrency Controller" as NodeA_LockMgr
end

participant "BATCH_CLUSTER_MEMBER\n(Heartbeat & Member Status)" as DB_MemberTable
participant "BATCH_LOCK\n(Pessimistic / Lease Lock)" as DB_LockTable
participant "BATCH_QUEUE\n(Distributed Job Queue)" as DB_QueueTable
end


participantgroup Batch Node B (Worker Node)
participant "Heartbeat & Queue Consumer" as NodeB_Consumer
participant "Batch Worker Execution Engine" as NodeB_Engine
end


group Phase 1: Node A Heartbeat, Leader Lock Acquisition & Job Registration

activate NodeA_LockMgr

activate DB_MemberTable

activate DB_LockTable

deactivate DB_LockTable

NodeA_Engine->DB_QueueTable: 4. INSERT INTO \nBATCH_QUEUE \n(Status: PENDING)

deactivate DB_QueueTable
end

group Phase 2: Node B Worker Heartbeat, Queue Polling & Execution

deactivate DB_MemberTable

activate NodeB_Engine

deactivate NodeB_Engine

NodeB_Consumer->DB_QueueTable: 9. UPDATE \nBATCH_QUEUE \nSET status='DONE'
activate DB_QueueTable
DB_QueueTable-->NodeB_Consumer: Job Status Updated \n(DONE)
deactivate DB_QueueTable
deactivate NodeB_Consumer
